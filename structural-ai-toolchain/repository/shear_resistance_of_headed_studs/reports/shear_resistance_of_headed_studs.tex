\noindent design resistance of a 19 mm headed stud welded through the deck, governed by stud shank failure or concrete crushing, checked against the design shear per stud.\par
\vspace{6pt}

\section*{\color{BrandPrimary}2.\ Calculation}
\noindent\textbf{Stud height factor}
\begin{align*}
\alpha &= 1 \;\mathrm{if}\; \frac{h_{sc}}{d} \geq 4,\;\; 0.2 \cdot \left(1 + \frac{h_{sc}}{d}\right) \;\mathrm{otherwise} \\
&= 1 \;\mathrm{if}\; \frac{100\ \mathrm{mm}}{19\ \mathrm{mm}} \geq 4,\;\; 0.2 \cdot \left(1 + \frac{100\ \mathrm{mm}}{19\ \mathrm{mm}}\right) \;\mathrm{otherwise} \\
&= \mathbf{1}
\end{align*}
\noindent\textbf{Design shear resistance of a headed stud — shank failure}
\begin{align*}
P_{Rd,1} &= \frac{0.8 \cdot \pi \cdot d^{2} \cdot f_{u}}{4 \cdot \gamma_{V}} \\
&= \frac{0.8 \cdot \pi \cdot \left(19\ \mathrm{mm}\right)^{2} \cdot 450\ \mathrm{MPa}}{4 \cdot 1.25} \\
&= \mathbf{81.66}\ \mathrm{kN}
\end{align*}
\noindent\textbf{Design shear resistance of a headed stud — concrete crushing}
\begin{align*}
P_{Rd,2} &= \frac{0.29 \cdot \alpha \cdot d^{2} \cdot \sqrt{E_{cm} \cdot f_{ck}}}{\gamma_{V}} \\
&= \frac{0.29 \cdot 1 \cdot \left(19\ \mathrm{mm}\right)^{2} \cdot \sqrt{33\ \mathrm{GPa} \cdot 30\ \mathrm{MPa}}}{1.25} \\
&= \mathbf{83.33}\ \mathrm{kN}
\end{align*}
\begin{align*}
P_{Rd} &= \min\left(P_{Rd,1}, P_{Rd,2}\right) \\
&= \min\left(81.66\ \mathrm{kN}, 83.33\ \mathrm{kN}\right) \\
&= \mathbf{81.66}\ \mathrm{kN}
\end{align*}
\begin{align*}
h_{eff} &= h_{sc} - 0.003\ \mathrm{m} \\
&= 100\ \mathrm{mm} - 0.003\ \mathrm{m} \\
&= \mathbf{97}\ \mathrm{mm}
\end{align*}
\begin{align*}
UR &= \frac{P_{Ed}}{P_{Rd}} \\
&= \frac{57.2\ \mathrm{kN}}{81.66\ \mathrm{kN}} \\
&= \mathbf{0.7005}
\end{align*}
\section*{\color{BrandPrimary}3.\ Verification}
\calccheckpass{$UR = 0.7005 \; \le \; 1$}\par\vspace{4pt}
