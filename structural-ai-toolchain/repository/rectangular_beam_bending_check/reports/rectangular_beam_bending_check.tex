{\LARGE\bfseries\color{BrandPrimary} Rectangular Beam Bending Check}\\[4pt]
\end{center}
\noindent Elastic bending stress verification for a simply supported rectangular timber beam under uniform load.\par
\vspace{6pt}
\section*{\color{BrandPrimary}1.\ Input Parameters}
\rowcolors{2}{BrandLight}{white}
\noindent\begin{tabular}{>{$}l<{$} r l p{7cm}}
\rowcolor{BrandPrimary}\textcolor{white}{\textbf{Symbol}} & \textcolor{white}{\textbf{Value}} & \textcolor{white}{\textbf{Unit}} & \textcolor{white}{\textbf{Description}} \\
b & 200 & mm & Section width \\
h & 400 & mm & Section depth \\
L & 5000 & mm & Span \\
q & 4.5 & kN/m & Distributed load (design) \\
f_{m} & 24 & MPa & Design bending strength \\
\end{tabular}\par\vspace{8pt}
\section*{\color{BrandPrimary}2.\ Calculation}
\noindent\textbf{Maximum bending moment}
\begin{align*}
M_{max} &= \frac{q \cdot \left(\frac{L}{1000}\right)^{2}}{8} \\
&= \frac{4.5 \cdot \left(\frac{5000}{1000}\right)^{2}}{8} \\
&= \mathbf{14.06}\ \mathrm{kNm}
\end{align*}
\noindent\textbf{Elastic section modulus}
\begin{align*}
W &= \frac{b \cdot h^{2}}{6} \\
&= \frac{200 \cdot 400^{2}}{6} \\
&= \mathbf{5.333e+06}\ \mathrm{mm}^{3}
\end{align*}
\noindent\textbf{Maximum bending stress}
\begin{align*}
\sigma_{m} &= M_{max} \cdot 1000000 \cdot \frac{1}{W} \\
&= 14.06 \cdot 1000000 \cdot \frac{1}{5.333e+06} \\
&= \mathbf{2.637}\ \mathrm{MPa}
\end{align*}
\section*{\color{BrandPrimary}3.\ Verification}
\calccheckpass{$\sigma_{m} = 2.637\ \mathrm{MPa} \; \le \; f_{m} = 24\ \mathrm{MPa}$}\par\vspace{4pt}
