\section*{\color{BrandPrimary}1.\ Input Parameters}

Span_{L} & 6 & m & Span L (m) \\
Line_{load,w} & 12 & kN/m & Line load w (kN/m) \\
Section_{modulus,Wel} & 324 & cm3 & Section modulus Wel (cm3) \\

\section*{\color{BrandPrimary}2.\ Calculation}
\noindent\textbf{Moment M (kNm)}
\begin{align*}
Moment_{M} &= \frac{Line_{load,w} \cdot Span_{L}^{2}}{8} \\
&= \frac{12 \cdot 6^{2}}{8} \\
&= \mathbf{54}\ \mathrm{kNm}
\end{align*}
\noindent\textbf{Bending stress sigma (MPa)}
\begin{align*}
Bending_{stress,\sigma} &= Moment_{M} \cdot 1000 \cdot \frac{1}{Section_{modulus,Wel}} \cdot factors_{b2} \\
&= 54 \cdot 1000 \cdot \frac{1}{324} \cdot 1 \\
&= \mathbf{166.7}\ \mathrm{MPa}
\end{align*}
\section*{\color{BrandPrimary}3.\ Verification}
\calccheckpass{$Bending_{stress,\sigma} = 166.7\ \mathrm{MPa} \; \le \; Yield_{fy} = 355\ \mathrm{MPa}$}\par\vspace{4pt}
